cmake_minimum_required(VERSION 3.22)
project(pokemon_plus_android C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -Wl,-z,max-page-size=16384")
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${PP_SDL_SOURCE_DIR}" sdl)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include")
if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/include/SDL2")
 file(CREATE_LINK "${PP_SDL_SOURCE_DIR}/include" "${CMAKE_CURRENT_BINARY_DIR}/include/SDL2" SYMBOLIC)
endif()
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../preview/lv_conf.h" CACHE STRING "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../preview/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
add_subdirectory(../managed_components/lvgl__lvgl lvgl)
target_include_directories(lvgl PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/../preview")
include(../preview/sources.cmake)
add_library(main SHARED ${PP_PREVIEW_SOURCES})
target_include_directories(main PRIVATE ../main ../core ../preview)
target_compile_definitions(main PRIVATE PP_APP_VERSION="${PP_APP_VERSION}")
target_compile_options(main PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(main PRIVATE lvgl SDL2 log android m)
