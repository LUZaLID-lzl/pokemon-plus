cmake_minimum_required(VERSION 3.16)
project(pokemon_plus_preview C)
set(CMAKE_C_STANDARD 11)
find_package(SDL2 REQUIRED)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(PP_LVGL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../managed_components/lvgl__lvgl" CACHE PATH "LVGL 9.5 source")
if(EXISTS "${PP_LVGL_DIR}/CMakeLists.txt")
 add_subdirectory(${PP_LVGL_DIR} lvgl)
else()
 include(FetchContent)
 FetchContent_Declare(lvgl URL https://codeload.github.com/lvgl/lvgl/tar.gz/refs/tags/v9.5.0)
 FetchContent_MakeAvailable(lvgl)
endif()
target_include_directories(lvgl PUBLIC ${SDL2_INCLUDE_DIRS})
include(sources.cmake)
add_executable(pokemon_plus ${PP_PREVIEW_SOURCES})
target_include_directories(pokemon_plus PRIVATE ../main ../core)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../CMakeLists.txt" PP_VERSION_LINE REGEX "set\\(PROJECT_VER")
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" PP_VERSION "${PP_VERSION_LINE}")
set(PP_APP_VERSION "${PP_VERSION}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../CMakeLists.txt")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../web/version.json.in" "${CMAKE_CURRENT_SOURCE_DIR}/../web/version.json" @ONLY)
target_compile_definitions(pokemon_plus PRIVATE PP_APP_VERSION="${PP_APP_VERSION}")
if(WIN32)
 enable_language(RC)
 string(REPLACE "." "," PP_VERSION_COMMAS "${PP_VERSION}")
 configure_file(windows.rc.in "${CMAKE_CURRENT_BINARY_DIR}/windows.rc" @ONLY)
 target_sources(pokemon_plus PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/windows.rc")
 set_target_properties(pokemon_plus PROPERTIES WIN32_EXECUTABLE TRUE OUTPUT_NAME PokemonPlus)
 if(TARGET SDL2::SDL2main)
  target_link_libraries(pokemon_plus PRIVATE SDL2::SDL2main)
 endif()
 target_link_libraries(pokemon_plus PRIVATE lvgl ${SDL2_LIBRARIES} bcrypt)
 if(MINGW)
  target_link_options(pokemon_plus PRIVATE -static-libgcc -static-libstdc++)
 endif()
else()
 target_link_libraries(pokemon_plus lvgl ${SDL2_LIBRARIES} m pthread)
endif()
target_compile_options(pokemon_plus PRIVATE -Wall -Wextra -Werror)

if(NOT WIN32)
# Exercise the actual ESP mmap/PSRAM branch against a real packed bank on host.
add_executable(pp_image_bank_test ../tests/test_image_bank.c ../main/image_assets.c ../main/scene_view.c ../core/scene_motion.c)
target_include_directories(pp_image_bank_test PRIVATE ../tests/mocks ../main ../core)
target_compile_definitions(pp_image_bank_test PRIVATE ESP_PLATFORM PP_ASSET_HOST_TEST PP_APP_VERSION="${PP_APP_VERSION}")
target_compile_options(pp_image_bank_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(pp_image_bank_test lvgl ${SDL2_LIBRARIES} m pthread z)

# Independent original48px PNG checks for centered native Flash pet storage.
add_executable(pp_pet_canvas_test ../tests/test_pet_canvas_sampling.c ../main/image_assets.c ../main/scene_view.c ../core/scene_motion.c)
target_include_directories(pp_pet_canvas_test PRIVATE ../tests/mocks ../main ../core)
